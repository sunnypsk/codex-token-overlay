cmake_minimum_required(VERSION 3.25)
project(CodexTokenOverlay VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(overlay_core core.cpp storage.cpp server.cpp coordinates.cpp claude.cpp)
target_include_directories(overlay_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} third_party)
target_compile_options(overlay_core PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(overlay_core PUBLIC bcrypt shell32 advapi32 shcore user32)
add_executable(overlay WIN32 app.cpp overlay.rc)
set_target_properties(overlay PROPERTIES OUTPUT_NAME "Codex Token Overlay")
target_compile_options(overlay PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(overlay PRIVATE overlay_core d2d1 dwrite comctl32 dwmapi ole32 oleacc oleaut32 uiautomationcore)
target_link_options(overlay PRIVATE /MANIFEST:NO)
add_executable(overlay_tests tests.cpp)
target_link_libraries(overlay_tests PRIVATE overlay_core)
target_compile_options(overlay_tests PRIVATE /W4 /utf-8 /EHsc)
enable_testing()
add_test(NAME native_core COMMAND overlay_tests)
